from sklearn.preprocessing import label_binarize
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.metrics import confusion_matrix, accuracy_score, f1_score, roc_curve, auc
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LogisticRegression

# Preprocess the 'reviewText' and 'summary' columns
def preprocess_data(training):
    # Preprocess the 'reviewText' and 'summary' columns
    training = preprocess_text(training)

    # Drop irrelevant columns
    dropped_training = training.drop(columns=['image', 'verified', 'reviewTime', 'reviewerName', 'asin', 
                                              'reviewerID', 'summary', 'category', 'vote', 'unixReviewTime'])

    # Combine 'reviewText' and 'summary' for more accuracy
    dropped_training['combined_text'] = training['reviewText'] + " " + training['summary']

    # Apply TF-IDF to 'combined_text' instead of 'reviewText'
    vectorizer = TfidfVectorizer(stop_words='english', max_df=0.3, min_df=5, 
                                 ngram_range=(1,2), sublinear_tf=True)

    X_tfidf = vectorizer.fit_transform(dropped_training['combined_text'])
    X_tfidf_dense = X_tfidf.toarray()
    tfidf_features = pd.DataFrame(X_tfidf_dense, columns=vectorizer.get_feature_names_out())

    # Drop unnecessary columns before concatenation
    dropped_training = dropped_training.drop(columns=['reviewText', 'summary', 'combined_text'], errors='ignore')
    training_cleaned = pd.concat([dropped_training, tfidf_features], axis=1)

    # Drop 'style' column if present
    if 'style' in training_cleaned.columns:
        training_cleaned = training_cleaned.drop(columns=['style'])

    for column in training_cleaned.select_dtypes(include=['object']).columns:
        training_cleaned[column] = training_cleaned[column].astype(str).str.replace("'", "").str.replace(":", "").str.strip()

    # Convert all remaining non-numeric columns to categorical 
    for column in training_cleaned.columns:
        if isinstance(training_cleaned[column], pd.Series) and training_cleaned[column].dtype == 'object':
            training_cleaned[column] = training_cleaned[column].astype('category').cat.codes

    return training_cleaned

    y_bin = label_binarize(y_test, classes=[1, 2, 3, 4, 5])
    fpr, tpr, roc_auc = {}, {}, {}

        fpr[i], tpr[i], _ = roc_curve(y_bin[:, i], model.predict_proba(X_test)[:, i])
        roc_auc[i] = auc(fpr[i], tpr[i])
